// ============ 0. Shared helpers =====================================================

// MODEL_PATH — change this to wherever you put the model/ folder. Do not hard-code a
// OneDrive path with your name in it if you plan to share the .pbix.
MODEL_PATH = "C:\VoltRelay\model"   // meta: edit this one line

// fnLoadCsv — one loader for every table, so delimiter/encoding/quoting is consistent.
// Text columns stay text; only keys and numerics are typed (avoid "123" → number on IDs).
let
    fnLoadCsv = (FileName as text, ColumnTypes as type) as table =>
    let
        Source   = Csv.Document ( File.Contents ( MODEL_PATH & "\" & FileName ),
                                  [ Delimiter = ",", Encoding = 65001, QuoteStyle = QuoteStyle.Csv ] ),
        Promoted = Table.PromoteHeaders ( Source, [ PromoteAllScalars = true ] ),
        Typed    = Table.TransformColumnTypes ( Promoted, ColumnTypes ),
        Trimmed  = Table.TransformColumns ( Typed,
                     List.Transform ( Table.ColumnNames ( Typed ), each { _, Text.Trim, type text } ),
                     null, MissingField.Ignore )
    in
        Trimmed
in
    fnLoadCsv

// FactRiderMonth — rider-month grain, the only valid source for churn.
let
    Source = fnLoadCsv ( "fact_rider_month.csv", type table [
        date_key = Int64.Type, month_start = type date, rider_id = text, attempts = Int64.Type,
